\begin{frame}
		\frametitle{Trees}
		\begin{center}
			\includegraphics[width=0.35\linewidth]{tree}		
		\end{center}
		
		\begin{itemize}
			\item A \textbf{tree} is a connected graph with no cycles.
			
			\item A \textbf{rooted tree} has a special node called the \textbf{root}.
			
			\item The root has no parent. All other nodes have exactly one parent.			
			
			\item \textbf{Leaves} are nodes with no children.
			
			
		\end{itemize}	
\end{frame}	

\begin{frame}
	\frametitle{Decision tree}
	\centering
	\includegraphics[width=0.6\linewidth]{decision_tree}		
	
	
	\begin{itemize}
		\item Internal nodes split the data based on feature values.
		\item Leaves assign a constant prediction (e.g. class label).
	\end{itemize}
\end{frame}	

\begin{frame}
	\frametitle{Decision tree - visualization}
	\centering
	\includegraphics[width=0.8\linewidth]{ess_tree_visualization.png}
	
	\begin{itemize}
		\item A decision tree models a piecewise constant function.		
		\item Can be used for both regression and classification.
	\end{itemize}
\end{frame}	

	
\begin{frame}
	\frametitle{Training a decision tree}
	
	\begin{itemize}
		\item Start with all data at the root.
		
		\item At each node, consider possible splits of the data.
		\pause
		
		\item Choose the \textbf{best split} according to a criterion:
		\begin{itemize}
			\item Regression: minimize residual sum of squares (RSS)
			\item Classification: minimize impurity (e.g. Gini index)
		\end{itemize}
		\pause
		\item Recursively repeat on resulting subsets.
		
		\item Stop when a stopping criterion is reached (e.g. small nodes).
		
		\pause
		\item Many variations exist (e.g. limiting depth, random feature selection).
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Gini index}
	\centering
	\includegraphics[width=0.6\linewidth]{gini_example}
	
	
	
	\begin{itemize}
		\item Gini index is defined as 
		$$G = 1 - \sum_{k=1}^{K} p_k^2$$
		where $p_k$ is the fraction of class $k$ among samples in the node.
		
		\item Equals the probability of misclassification (if labels are assigned according to class proportions).
		\item Used to train classification trees.
	\end{itemize}
	
\end{frame}

	
\begin{frame}
	\frametitle{Bias-variance trade-off}
	\begin{itemize}    
		\item \textbf{Shallow trees}: low variance, high bias (may miss complex patterns).    
		\item \textbf{Deep trees}: low bias, high variance (prone to overfitting).    
		
		\item The trade-off can be controlled by model parameters (e.g. tree depth, minimum node size).
		\pause        
		\item Can we get the best of both worlds?
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Ensembling methods}
	\begin{itemize}    
		\item Idea: train multiple models, average their predictions.
		\pause
		\item \textbf{Bagging} (Bootstrap aggregating):
		\begin{itemize}
			\item Sample different training sets from the original data.
			\item Train \textit{high variance, low bias} models (deep trees) on these sets and average them.
			\item Exploits independence between models.
		\end{itemize}
	
	\pause
	\item \textbf{Boosting}:
	\begin{itemize}
		\item Sequentially train \textit{low variance, high bias} models (shallow trees).
		\item Subsequent models learn to fix the mistakes of the previous ones.
		\item Exploits dependence between models.
	\end{itemize}
	
		
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Bootstrapping}
	
	\centering
	\includegraphics[width=0.65\linewidth]{bootstrapping}
	
	\begin{itemize}
		\item Resampling from the original training set (with replacement).
		\begin{itemize}
			\item Same number of observations.
			\item Some observations may be sampled multiple times.
			\item Some observations may not be sampled at all.
		\end{itemize}
		
		\item Each resampling produces a slightly different version of the original data.
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Random Forest} 
	
	\centering
	\includegraphics[width=0.35\linewidth]{forest}
	
	\begin{itemize}    
		\item Uses bootstrapping to generate many training datasets from the original data.
		\item Train deep trees on these datasets and average their predictions.
		\begin{itemize}
			\item Reduces variance while keeping low bias.
		\end{itemize}	
		\pause	
		\item Additional randomness: at each split, consider only a random subset of features.
		\begin{itemize}
			\item Reduces correlation between trees.
			\item May also prevent trees from getting stuck in local optima during training.
		\end{itemize}			
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Out-of-bag (OOB) samples} 
	\centering
	\includegraphics[width=0.75\linewidth]{oob}
	\begin{itemize}
		\item For every tree, observations not sampled by bootstrap (out-of-bag, OOB) may be used as a test set.
		\item For each observation, aggregate predictions from the trees where it was out-of-bag.
		\begin{itemize}
			\item Provides an estimate of the test error of the whole random forest.
		\end{itemize}
	\end{itemize}	
\end{frame}

\begin{frame}
	\frametitle{Feature importance} 
	\centering
	\includegraphics[width=0.75\linewidth]{feature_importance}
	\begin{itemize}
		\item Randomly permute a feature in OOB samples (breaking its relationship with the outcome) and measure the decrease in prediction performance.
	\end{itemize}	
\end{frame}

\begin{frame}
	\frametitle{Boosting} 
	
\begin{itemize}
	\item \textbf{AdaBoost} (adaptive boosting): 
	\begin{itemize}
		\item Sequentially train very shallow trees (so-called \textit{decision stumps}).
		\item Models aim to minimize a weighted loss (samples have weights).
		\item Misclassified samples get higher weight, so future models focus on them.
		\item Combine models using a weighted average (better models get higher weight).
	\end{itemize}
\pause

\item \textbf{Gradient boosting} (e.g. XGBoost):
\begin{itemize}
	\item Similar idea: sequentially add models to correct previous errors.
	\item Fit new models to the gradients of the loss (for RSS, this corresponds to residuals).
	\item Uses small trees (deeper than decision stumps).
\end{itemize}
\end{itemize}

\end{frame}
