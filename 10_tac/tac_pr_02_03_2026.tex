\begin{frame}
	\frametitle{RNA metabolism as a dynamical system}
	\pause
	 For a given gene, following molecules are present in the system: 
	 \pause 
	 \begin{itemize} 
	 	\item Nascent and mature transcripts: $T_{\mathrm{nascent}}, \; T_{\mathrm{mature}}$. \pause
	 	\item Nascent and finished (but unspliced) introns: ${I}_{\mathrm{nascent}}, {I}_{\mathrm{unspliced}}$. 
	 	\pause 
 	\end{itemize}
	

	
	\vspace{0.5em}
	
	The dynamics may be described by an ODE system:	\pause
	\begin{align*}
		\dot{T}_{\mathrm{nascent}} 
		&= k_{\mathrm{init}}
		- k_{\mathrm{elong}} \cdot T_{\mathrm{nascent}} \cdot L_{\mathrm{gene}}^{-1} \\		
		\dot{{I}}_{\mathrm{nascent}} 
		&= k_{\mathrm{init}}
		- k_{\mathrm{elong}} \cdot {I}_{\mathrm{nascent}} \, \cdot L_{\mathrm{intron}}^{-1} \\		
		\dot{T}_{\mathrm{mature}} 
		&= k_{\mathrm{elong}} \cdot T_{\mathrm{nascent}} \, \cdot L_{\mathrm{gene}}^{-1}
		- k_{\mathrm{deg}} \cdot T_{\mathrm{mature}} \\		
		\dot{{I}}_{\mathrm{unspliced}} 
		&= k_{\mathrm{elong}} \cdot {I}_{\mathrm{nascent}} \, \cdot L_{\mathrm{intron}}^{-1}
		- k_{\mathrm{splice}} \cdot {I}_{\mathrm{unspliced}}
	\end{align*}
where we have
\begin{itemize}
	\item Kinetic rates: $[k_{\mathrm{init}}] = \mathrm{polymerases}\cdot \mathrm{s}^{-1}$, $[k_{\mathrm{elong}}]= \mathrm{bp}\cdot \mathrm{s}^{-1}$, 
	$[k_{\mathrm{splice}}] = \mathrm{s}^{-1}$, $[k_{\mathrm{deg}}] = \mathrm{s}^{-1}$
	\item Gene and intron lenghts: $[L_{\mathrm{gene}}] = [L_{\mathrm{intron}}] = \mathrm{bp}$
\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Steady state solution}
	\pause
	We may solve the ODE system for steady state solution:
	\begin{align*}
		{T}_{\mathrm{nascent}} 
		&= \frac{k_{\mathrm{init}}}{k_{\mathrm{elong}}} \cdot L_{\mathrm{gene}}  \\[6pt]	
		{I}_{\mathrm{nascent}} 
		&= \frac{k_{\mathrm{init}}}{k_{\mathrm{elong}}} \cdot L_{\mathrm{intron}} \\[6pt]	
		{T}_{\mathrm{mature}} 
		&= \frac{k_{\mathrm{init}}}{k_{\mathrm{deg}}} \\[6pt]		
		{I}_{\mathrm{unspliced}} 
		&= \frac{k_{\mathrm{init}}}{k_{\mathrm{splice}}}
	\end{align*}

\pause
\begin{itemize}
	\item The steady state quantities depends only on the ratios of kinetic rates.
	\pause
	\item Therefore, scaling all kinetic rate by some constant $c \in \mathbb{R}_+$ does not change the steady state.
\end{itemize} 
	
\end{frame}


\begin{frame}
	\frametitle{Non-identifiability of kinetic rates from steady state}
	\pause
	\begin{center}
		\includegraphics[width=9cm]{invariance_to_scaling.png}
	\end{center}
	\pause
	\begin{itemize}
		\item Only ratios of kinetic rates, e.g., $\frac{k_{\mathrm{elong}}}{k_{\mathrm{deg}}}$, are identifiable. \pause
		\item When comparing samples (treatment vs. control), only differences of LFC can be estimated and statistically tested. \pause
		\item 
		E.g., we may estimate the value of 
		$$
		\mathrm{LFC}(k_{\mathrm{elong}}) - \mathrm{LFC}(k_{\mathrm{deg}})
		$$
		and test the hypothesis
		$$
		\mathrm{H}_{0}: \quad \mathrm{LFC}(k_{\mathrm{elong}}) - \mathrm{LFC}(k_{\mathrm{deg}}) = 0
		$$
	\end{itemize}
\end{frame}


\begin{frame}
	\frametitle{Non-identifiability is a common problem}
	\pause
	\begin{itemize}
		\item Consider an autonomous dynamical system with state vector $\boldsymbol{x}$, and with dynamics
		$$
		\dot{\boldsymbol{x}} = f(\boldsymbol{x})
		$$
		\pause
		\item It holds that
		$$	
		f(\boldsymbol{x}) = 0 \Leftrightarrow c \cdot f(\boldsymbol{x}) = 0 \quad  \forall c \in \mathbb{R}_+
		$$
		\pause 
		\item Therefore, steady-state data cannot distinguish $f(\boldsymbol{x})$ from $c \cdot f(\boldsymbol{x})$.  
		Consequently, only ratios of rates are identifiable.
		\pause
		\item This type of structural non-identifiability is common in RNA metabolism models.
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Dealing with structural non-identifiability}
	\pause
	\begin{itemize}
		\item Use orthogonal experiment to measure at least one kinetic rate independently.
		\pause
		\item Perturb the biological system to obtain non-steady-state observations.
		\pause
		\begin{itemize}
			\item E.g., inhibiting/releasing transcription, and/or using labeled nucleotides. 
		\end{itemize}
	
	\end{itemize} 
	
	
\end{frame}


\begin{frame}
	\frametitle{Compositional non-identifiability in RNA-seq}
	\pause
	\begin{itemize}
		\item RNA-seq data are compositional: we observe proportions within samples, not absolute abundances. This introduces an additional source of non-identifiability. \pause
	 \item Let
	 $$
	 S_i = \{ \text{RNA molecules in sample } i\}
	 $$
	 We are typically interested in some subset $A_i \subset S_i$, and we aim to compare its size between samples, i.e. $\frac{|A_2|}{|A_1|}$.
	 \pause
	 \item With RNA-seq, we observe the number of reads $r_i$ corresponding to the molecules of interest, which scales as
	 $$
	 r_i \propto \frac{|A_i|}{|S_i|} = \frac{|A_i|}{|A_i|+|A_i^c|}
	 $$
	 where $A^c_i$ is a complement of $A_i$ in $S_i$.
	 \end{itemize}	
\end{frame}

	\end{itemize}	
	
\end{frame}

\begin{frame}
	\frametitle{Conclusion \& open questions}
	\pause
	\begin{itemize}
		\item Estimating changes in RNA kinetic rates from total RNA-seq data poses two non-identifiability problems:
		\pause
		\begin{itemize}
			\item Structural non-identifiability: the underlying dynamical system cannot be identified from steady-state data.
			\pause
			\item Compositional non-identifiability: arises because RNA-seq measures only relative abundances within a sample.
		\end{itemize}
	\pause
		\item To overcome these issues, we need either additional measurements, and/or different types of data than total RNA-seq.
	\end{itemize}	
	\pause

\end{frame}
